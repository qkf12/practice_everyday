select   专项练习
		select 列名，列名 from 表明；
		select 列名 as 别名，列名 as 别名 from 表明；
		select 列名 别名，列名 别名 from 表名；

		select * from 表名 as 别名；		给表起名是为了合并表格的时候用
		select * from 表名 别名；

		
		select  distinct * from 表名； 去掉查询时的重复数据

		2   where 语句
			select * from 表名 where  条件 and 条件；
			<>  是不等于  同等于 !=
			select * from 表名  not  where 条件；   相反的条件  跟!= 很像
				
		3   模糊查询语句
			select * from 表名 where name like "张%";  查找以张开头的名字
			select * from 表名 where name like "%张";  查找以张结尾的名字

			select * from 表名 where name like "张_";   查找以张开头且名字是两位的
			select * from 表名 where name like "张__";   查找以张开头且名字是三位的




			select * from students where hometown in ("北京"，"上海"，"广东");
			select * from students where age between  25 and 30; 大于25而且小于 30的

		   

		4   空判断
			select * from students where card is null;
			select * from students where card is not null;
			
		5   排序                           where    在order by  之前
			select * from students order by  字段（列） asc；(默认) 从小到大
			select * from students order by  age  desc;  从大到小排列

			select * from students order by age desc, student_id desc;  年龄从大到小排列  如果年龄相同 从学号从大到小排列


		6   函数
			select count(*) from students;  返回多少条记录的
			select count(distinct class) from students;   去除重复值之后 有多少条记录
			select max(age) from student where sex = "女"; 查询最大的年龄
			同理  min  sun  avg（平均值）

			avg   平均值忽略  null   分母相对应的也减一；

		7   分组
			select * from students where class = "1班"  group by  sex；
	
	一个句子  一般是   整句  + 条件 +  分组 + 排序
		排序必须是在最后
		
			分组之后才能的数据筛选  having   having  可以使用聚合函数
				select  class  from students  group  by  class  having  count(*)> 3;		查看  班级人数大于3人的 班级名称


		8   分页
			select * from students limit 0,3;
			select * from students limit 3;

		9  内连接
			select * from 表1 inner join 表2 on 表1.字段 = 表2.字段
		where  条件； 					
			隐式内 连接
					select * from 表1，表2 where 表1.字段 = 表2.字段；

			多表连接
				select * from 表1 inner join 表2  on 表1.字段 = 表2.字段
			inner join 表3 on 表2.字段 = 表3.字段；
		10 左连接    （把inner 换成left）同理右连接 right

		11 自关联   (把自己看成两张表   用别名关联一下)  同一张表
			select * from 表1 b1  inner  join  表1 b2
		on  b1.字段 = b2.字段   where  条件;


		12  子查询  （两个select   主查询  ）
			例子
				select * from students  where  age  > ( select avg(age) from students);			可以多表查询  
			  不能用having   分组之后才能用


			标量子查询    返回的是一个结果
